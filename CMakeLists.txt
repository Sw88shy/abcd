cmake_minimum_required(VERSION 3.16)
project(usb_people_detector LANGUAGES CXX)

include(CTest)
option(BUILD_DETECTOR "Build the camera application (requires OpenCV and NCNN)" ON)
option(ENABLE_MOTORS "Build Linux L298N GPIO support using lgpio" OFF)
add_library(person_navigation src/person_navigation.cpp)
target_compile_features(person_navigation PUBLIC cxx_std_17)
target_include_directories(person_navigation PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
if(BUILD_TESTING)
    find_package(Threads REQUIRED)
    add_executable(navigation_tests tests/navigation_tests.cpp)
    target_link_libraries(navigation_tests PRIVATE person_navigation)
    add_test(NAME navigation COMMAND navigation_tests)
    add_executable(motor_tests tests/motor_tests.cpp src/motor_driver.cpp)
    target_include_directories(motor_tests PRIVATE tests/fake_gpio)
    target_compile_definitions(motor_tests PRIVATE HAVE_LGPIO)
    target_link_libraries(motor_tests PRIVATE person_navigation Threads::Threads)
    add_test(NAME motor_shutdown COMMAND motor_tests)
endif()

if(BUILD_DETECTOR)
    find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc videoio highgui)
    find_package(ncnn CONFIG REQUIRED)
    find_package(Threads REQUIRED)
    add_executable(people_detector src/main.cpp src/yolo_detector.cpp src/motor_driver.cpp)
    target_compile_features(people_detector PRIVATE cxx_std_17)
    target_include_directories(people_detector PRIVATE ${OpenCV_INCLUDE_DIRS})
    target_link_libraries(people_detector PRIVATE ${OpenCV_LIBS} ncnn person_navigation Threads::Threads)
    if(ENABLE_MOTORS)
        if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
            message(FATAL_ERROR "ENABLE_MOTORS requires Linux on the Raspberry Pi")
        endif()
        find_path(LGPIO_INCLUDE_DIR lgpio.h)
        find_library(LGPIO_LIBRARY lgpio)
        if(NOT LGPIO_INCLUDE_DIR OR NOT LGPIO_LIBRARY)
            message(FATAL_ERROR "Install liblgpio-dev to build motor support")
        endif()
        target_include_directories(people_detector PRIVATE ${LGPIO_INCLUDE_DIR})
        target_link_libraries(people_detector PRIVATE ${LGPIO_LIBRARY})
        target_compile_definitions(people_detector PRIVATE HAVE_LGPIO)
    endif()
    if(MSVC)
        target_compile_options(people_detector PRIVATE /W4)
    else()
        target_compile_options(people_detector PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endif()
